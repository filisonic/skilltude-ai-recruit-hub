# Render Environment Variables
# Copy these to Render's Environment Variables section

# ===== REQUIRED - FILL THESE IN =====

NODE_ENV=production
PORT=10000

# Database (Get from Hostinger phpMyAdmin)
DB_HOST=your_hostinger_mysql_host
DB_USER=your_database_username
DB_PASSWORD=your_database_password
DB_NAME=your_database_name
DB_PORT=3306

# Email (Gmail App Password)
SMTP_HOST=smtp.gmail.com
SMTP_PORT=587
SMTP_SECURE=false
SMTP_USER=your_email@gmail.com
SMTP_PASS=your_gmail_app_password
EMAIL_FROM=noreply@skilltude.com

# Admin Credentials (Create secure ones)
ADMIN_EMAIL=admin@skilltude.com
ADMIN_PASSWORD=create_a_secure_password_here

# JWT Secret (Generate random string - at least 32 characters)
JWT_SECRET=generate_a_long_random_string_here_at_least_32_chars

# File Upload
UPLOAD_DIR=./uploads
MAX_FILE_SIZE=5242880

# Frontend URL (Your Hostinger domain - for CORS)
FRONTEND_URL=https://your-domain.com

# Rate Limiting
RATE_LIMIT_WINDOW_MS=900000
RATE_LIMIT_MAX_REQUESTS=100

# ===== HOW TO GET THESE VALUES =====

# DB_HOST: 
# - Go to Hostinger hPanel → Databases → phpMyAdmin
# - Look for "Server" or "Host" (usually something like: mysql123.hostinger.com)

# DB_USER, DB_PASSWORD, DB_NAME:
# - In Hostinger hPanel → Databases
# - You created these when setting up MySQL

# SMTP_PASS (Gmail App Password):
# 1. Go to myaccount.google.com/security
# 2. Enable 2-Step Verification
# 3. Search for "App Passwords"
# 4. Generate password for "Mail"
# 5. Copy the 16-character password

# JWT_SECRET:
# Generate a random string. You can use this command:
# node -e "console.log(require('crypto').randomBytes(32).toString('hex'))"

# CORS_ORIGIN:
# Your Hostinger website URL (e.g., https://skilltude.com)
